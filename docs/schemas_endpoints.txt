# Schemas por endpoint - Alejandria (formato simple para IA)
# Cada bloque: METHOD PATH
# auth_required: yes|no
# request: SchemaName
#   field: type (constraints)
# response: SchemaName (or primitive)
#   field: type

---
GET /
auth_required: no
request: none
response: object
  message: str

---
POST /auth/signup
auth_required: no
request: UserAdminBase
  username: str (min_length=3, max_length=50)
  password: str (min_length=6, max_length=128)
  role_id: int (gt=0)
response: UserAdminResponse
  id: int
  username: str
  role_id: int
  is_active: bool

---
POST /auth/token
auth_required: no
request: form-data
  username: str
  password: str
response: Token
  access_token: str
  token_type: str
notes: implementation returns access_token as username (string)

---
GET /users/
auth_required: yes
request: none
response: list[UserResponse]
  id: int
  username: str
  role_id: int
  total_publications: int
  role: RoleResponse
    name: str
    description: str | None
    max_publications: int
  created_at: datetime
  is_active: bool

---
POST /users/
auth_required: yes
request: UserCreate
  username: str
  role_id: int
  is_active: bool
response: UserResponse
  id: int
  username: str
  role_id: int
  total_publications: int
  role: RoleResponse
    name: str
    description: str | None
    max_publications: int
  created_at: datetime
  is_active: bool

---
GET /users/{user_id}/movements
auth_required: yes
request: path user_id: int
response: list[MovementMini]
  publication: PublicationMini
    id: int
    name: str
    code: str
  quantity: int
  movement_type: str
  timestamp: datetime

---
PUT /users/{user_id}
auth_required: yes
request: UserUpdate
  username: str
  role_id: int
  is_active: bool
response: UserResponse (same as above)

---
